2. Sample Playback Engine

Implementation Plan:

```cmajor
// New Sample Engine Module
processor SamplePlaybackEngine
{
    input stream float32 in;
    input event int32 sampleIndex;
    output stream float32 out;
    
    // Memory-mapped sample buffers
    audio_buffer[72] phonemeSamples;  // 24-bit/96kHz WAV files
    
    // Sample playback modes
    enum PlaybackMode {
        Exact,      // Natural phoneme duration
        Sustain,    // Looped vowel region
        Staccato    // 50% natural duration with fade
    }
    
    // Zero-latency playback with pitch preservation
    void main()
    {
        loop
        {
            // Sample-accurate playback
            let sample = phonemeSamples[sampleIndex].read();
            out <- sample * envelope;
            advance();
        }
    }
}
```

1. AI-Assisted Phoneme Sequencing

Concept:

```python
class AI_PhonemeSequencer:
    """
    Uses machine learning to generate authentic
    ancient Egyptian phrases from text input.
    """
    
    def __init__(self):
        self.model = load_model("egyptian_phoneme_generator")
        self.vocabulary = load_vocabulary("middle_egyptian")
    
    def generate_phrase(self, text):
        """
        Convert modern text to ancient Egyptian phonemes
        with authentic co-articulation patterns.
        """
        # 1. Tokenize input
        tokens = self.tokenize(text)
        
        # 2. Map to phoneme sequences
        phonemes = [self.model.predict(token) for token in tokens]
        
        # 3. Apply co-articulation rules
        with_coarticulation = self.apply_coarticulation(phonemes)
        
        # 4. Generate prosody
        prosody = self.generate_prosody(with_coarticulation)
        
        return prosody
    
    def learn_from_manuscripts(self, corpus):
        """
        Train on actual Egyptian texts to learn
        authentic speech patterns.
        """
        self.model.train(corpus)
```

namespace hieroglyph

    //--------------------------------------------------------------------------
    // Sample Playback Processor
    //--------------------------------------------------------------------------
    processor SamplePlayback
    {
        input stream float32 in;
        input event int32 sampleIndex;
        output stream float32 out;

        // Sample buffers (loaded at runtime)
        audio_buffer[72] phonemeSamples;
        
        // Playback state
        int32 currentSample = -1;
        float32 playhead = 0.0f;
        float32 playbackRate = 1.0f;
        float32 sampleDuration = 0.0f;
        
        // Envelope for smooth playback
        float32 envelope = 0.0f;
        float32 attackMs = 5.0f;
        float32 releaseMs = 20.0f;

        event sampleIndex (int32 idx)
        {
            if (idx >= 0 && idx < 72)
            {
                currentSample = idx;
                playhead = 0.0f;
                sampleDuration = phonemeSamples[idx].duration;
            }
        }

        void main()
        {
            loop
            {
                if (currentSample >= 0 && playhead < sampleDuration)
                {
                    // Read sample with interpolation
                    let sample = readSample(currentSample, playhead);
                    
                    // Apply envelope
                    let env = calculateEnvelope();
                    out <- sample * env;
                    
                    // Advance playhead
                    playhead += playbackRate;
                    
                    // Handle end of sample
                    if (playhead >= sampleDuration)
                    {
                        currentSample = -1;
                    }
                }
                else
                {
                    out <- 0.0f;  // Silence when no sample
                }
                
                advance();
            }
        }

        // Sample reading with linear interpolation
        float32 readSample(int32 sampleIdx, float32 position)
        {
            let idx = int32(position);
            let frac = position - float32(idx);
            
            let s1 = phonemeSamples[sampleIdx].read(idx);
            let s2 = phonemeSamples[sampleIdx].read(idx + 1);
            
            return s1 * (1.0f - frac) + s2 * frac;
        }

        // Envelope calculation
        float32 calculateEnvelope()
        {
            let attackSamples = attackMs * 0.001f * float32(processor.frequency);
            let releaseSamples = releaseMs * 0.001f * float32(processor.frequency);
            
            if (playhead < attackSamples)
            {
                return playhead / attackSamples;
            }
            else if (playhead > sampleDuration - releaseSamples)
            {
                return (sampleDuration - playhead) / releaseSamples;
            }
            
            return 1.0f;
        }
    }

    //--------------------------------------------------------------------------
    // Enhanced Phoneme Voice with Sample Support
    //--------------------------------------------------------------------------
    graph PhonemeVoiceV11 [[ main ]]
    {
        input event std::midi::Message midiIn;
        output stream float32 out;

        node glottal = GlottalExcitation;
        node noise = NoiseExcitation;
        node bank = FormantFilterBank;
        node env = AmpEnvelope;
        node coartic = CoarticulationController;
        node ritual = RitualOvertoneLayer;
        node sampler = SamplePlayback;  // NEW: sample playback
        
        // Hybrid morph control
        input event float32 morphAmount;  // 0 = synth, 1 = sample

        bool hasCurrent = false;
        float32 mixCoeff = 1.0f;

        event midiIn (std::midi::Message m)
        {
            let msg = std::midi::getMessageDescription (m);

            if (msg.isNoteOn())
            {
                let idx = int (msg.note) % phonemeTable.size;
                let newPhoneme = phonemeTable.at (idx);

                mixCoeff = newPhoneme.voicedMix;
                glottal.noteFrequency <- std::notes::noteToFrequency (msg.note);
                ritual.fundamentalFreq <- std::notes::noteToFrequency (msg.note);

                // Route to co-articulation
                coartic.formantTargets <- newPhoneme.formants;
                coartic.qTargets <- newPhoneme.qValues;
                coartic.transitionMs <- hasCurrent ? 15.0f : 0.0f;

                {

                }

                hasCurrent = true;
                env.gate <- msg.velocity;
            }
            else if (msg.isNoteOff())
            {
                env.gate <- 0.0f;
            }
            else if (msg.isPolyphonicAftertouch())

                mixCoeff = mixCoeff + (emph - mixCoeff) * 0.5f;
                env.gate <- max (env.gate, emph * 0.8f);
                ritual.overtoneMix <- emph;
            }
        }

        event morphAmount (float32 m) { morph = m; }

        connection
        {
            // Synth chain
            glottal.out * mixCoeff -> bank.in;
            noise.out * (1.0f - mixCoeff) -> bank.in;
            bank.out -> ritual.in;
            
            // Sample chain
            sampler.out -> ritual.in;
            
            // Hybrid morph
            ritual.out * (1.0f - morph) + sampler.out * morph -> out;
            
            // Envelope
            env.out -> out;
            
            // Co-articulation
            coartic.formantOutputs -> bank.formantFreqs;
            coartic.qOutputs -> bank.formantQs;
        }

    //--------------------------------------------------------------------------
    // SAMPLE PLAYBACK ENGINE
    //--------------------------------------------------------------------------
    processor SamplePlaybackEngine
    {
        input stream float32 in;
        input event int32 sampleIndex;
        input event float32 morphAmount;
        output stream float32 out;

        audio_buffer[72] phonemeSamples;
        int32 currentSample = -1;
        float32 playhead = 0.0f;
        float32 sampleDuration = 0.0f;
        float32 morph = 0.35f;

        event sampleIndex (int32 idx)
        {
            if (idx >= 0 && idx < 72)
            {
                currentSample = idx;
                playhead = 0.0f;
                sampleDuration = phonemeTable[idx].durationMs * 0.001f;
            }
        }

        event morphAmount (float32 m) { morph = m; }

        void main()
        {
            loop
            {
                if (currentSample >= 0 && playhead < sampleDuration)
                {
                    let sample = readSample(currentSample, playhead);
                    let synth = in;
                    out <- synth * (1.0f - morph) + sample * morph;
                    playhead += 1.0f / float32(processor.frequency);
                }
                else
                {
                    out <- in;
                }
                advance();
            }
        }

        float32 readSample(int32 idx, float32 pos)
        {
            let sampleIdx = int32(pos * float32(processor.frequency));
            return phonemeSamples[idx].read(sampleIdx);
        }
    }

    //--------------------------------------------------------------------------
    // ENHANCED CO-ARTICULATION WITH COMPONENT PHONEMES
    //--------------------------------------------------------------------------
    processor EnhancedCoarticulation
    {
        input event int32[3] componentPhonemes;
        input event float32 transitionMs;
        output stream float32[5] formantOutputs;
        output stream float32[5] qOutputs;

        float32[5] currentFormants;
        float32[5] targetFormants;
        float32[5] currentQs;
        float32[5] targetQs;
        float32 progress = 1.0f;
        float32 transitionSamples = 480.0f;

        event componentPhonemes (int32[3] comps)
        {
            // Get formants from first component
            let p = phonemeTable[comps[0]];
            targetFormants = p.formants;
            targetQs = p.qValues;
            progress = 0.0f;
        }

        event transitionMs (float32 ms)
        {
            transitionSamples = ms * 0.001f * float32(processor.frequency);
        }

        void main()
        {
            loop
            {
                if (progress < 1.0f)
                {
                    progress = min(progress + 1.0f / transitionSamples, 1.0f);
                    let t = progress;
                    // Bézier interpolation
                    for (wrap<5> i)
                    {
                        currentFormants[i] = currentFormants[i] + (targetFormants[i] - currentFormants[i]) * t;
                        currentQs[i] = currentQs[i] + (targetQs[i] - currentQs[i]) * t;
                    }
                }
                
                formantOutputs <- currentFormants;
                qOutputs <- currentQs;
                advance();
            }
        }
    }

    //--------------------------------------------------------------------------
    // RITUAL LAYER WITH SAMPLE SUPPORT
    //--------------------------------------------------------------------------
    processor RitualLayer
    {
        input stream float32 in;
        input event float32 fundamentalFreq;
        input event float32 overtoneMix;
        input event float32 shimmerRate;
        input event float32 emphasis;
        output stream float32 out;

        float32 f0 = 220.0f;
        float32 overtone = 0.0f;
        float32 shimmer = 2.0f;
        float32 emph = 0.5f;
        float32[8] phase;
        float32 shimmerPhase = 0.0f;

        event fundamentalFreq (float32 f) { f0 = f; }
        event overtoneMix (float32 m) { overtone = m; }
        event shimmerRate (float32 r) { shimmer = r; }
        event emphasis (float32 e) { emph = e; }

        void main()
        {
            loop
            {
                float32 sum = 0.0f;
                for (wrap<8> h)
                {
                    let harmonic = float32(int(h) + 2);
                    let inc = (f0 * harmonic) / float32(processor.frequency);
                    phase[h] += inc;
                    if (phase[h] >= 1.0f) phase[h] -= 1.0f;

                    let am = 0.5f + 0.5f * sin(twoPi * (shimmerPhase + float32(int(h)) * 0.07f));
                    sum += sin(twoPi * phase[h]) * (1.0f / harmonic) * am;
                }

                shimmerPhase += shimmer / float32(processor.frequency);
                if (shimmerPhase >= 1.0f) shimmerPhase -= 1.0f;

                let overtones = sum * 0.35f * emph;
                out <- in * (1.0f - overtone) + (in + overtones) * overtone;
                advance();
            }
        }
    }

    //--------------------------------------------------------------------------
    // ENHANCED PHONEME VOICE
    //--------------------------------------------------------------------------
    graph PhonemeVoiceV11 [[ main ]]
    {
        input event std::midi::Message midiIn;
        output stream float32 out;

        node glottal = GlottalExcitation;
        node noise = NoiseExcitation;
        node bank = FormantFilterBank;
        node env = AmpEnvelope;
        node sampler = SamplePlaybackEngine;
        node ritual = RitualLayer;
        node coartic = EnhancedCoarticulation;

        bool hasCurrent = false;
        float32 mixCoeff = 1.0f;
        float32 morph = 0.35f;

        event midiIn (std::midi::Message m)
        {
            let msg = std::midi::getMessageDescription(m);

            if (msg.isNoteOn())
            {
                let idx = int(msg.note) % phonemeTable.size;
                let p = phonemeTable[idx];

                mixCoeff = p.voicedMix;
                glottal.noteFrequency <- std::notes::noteToFrequency(msg.note);
                ritual.fundamentalFreq <- std::notes::noteToFrequency(msg.note);

                // Co-articulation
                if (p.isBiliteral || p.isTriliteral)
                {
                    coartic.componentPhonemes <- p.componentPhonemes;
                    coartic.transitionMs <- hasCurrent ? 15.0f : 0.0f;
                }

                // Sample playback
                if (p.sampleIndex >= 0)
                {
                    sampler.sampleIndex <- p.sampleIndex;
                }

                hasCurrent = true;
                env.gate <- msg.velocity;
            }
            else if (msg.isNoteOff())
            {
                env.gate <- 0.0f;
            }
            else if (msg.isPolyphonicAftertouch())
            {
                let emph = msg.aftertouchValue / 127.0f;
                mixCoeff = mixCoeff + (emph - mixCoeff) * 0.5f;
                env.gate <- max(env.gate, emph * 0.8f);
                ritual.emphasis <- emph;
            }
        }

        connection
        {
            glottal.out * mixCoeff -> bank.in;
            noise.out * (1.0f - mixCoeff) -> bank.in;
            bank.out -> ritual.in;

            sampler.out * morph -> ritual.in;

            ritual.out * env.out -> out;

            coartic.formantOutputs -> bank.formantFreqs;
            coartic.qOutputs -> bank.formantQs;
        }
    }

#!/usr/bin/env python3
"""
Generate sample variations for the Hieroglyph Phonogram Engine v1.1
Adds multiple takes, tempos, and acoustic treatments
"""

import numpy as np
from scipy.io import wavfile
from scipy.signal import lfilter, butter
import os

SR = 44100
OUT = "/home/workdir/artifacts/samples_variations"
os.makedirs(OUT, exist_ok=True)

# Reuse existing phoneme data
from generate_full_72 import uni, seg, coart, rev, write

# Generate variations for key phonemes
variations = {
    # (phoneme, variations to generate)
    "ꜥnḫ": [("slow_temple", 0.8, 0.5), ("fast_ritual", 1.2, 0.3), ("dark_oracle", 0.7, 0.6)],
    "nfr": [("bright_sun", 1.1, 0.2), ("warm_hymn", 0.9, 0.4), ("mystic", 0.85, 0.5)],
    "ḫpr": [("transform", 1.0, 0.4), ("becoming", 0.9, 0.5), ("emergence", 1.1, 0.3)],
    "snb": [("healing", 0.95, 0.4), ("healthy", 1.05, 0.3), ("vitality", 1.15, 0.2)],
    "wbn": [("sunrise", 0.9, 0.5), ("shining", 1.1, 0.3), ("dawn", 0.85, 0.6)]
}

print("=== Generating Variations ===")
for phoneme, variations_list in variations.items():
    for var_name, speed, reverb_mix in variations_list:
        # Use base phoneme data
        if phoneme == "ꜥnḫ":
            keys = ["ꜥ", "n", "ḫ"]
            durs = [0.20/speed, 0.16/speed, 0.18/speed]
            f0s = [100, 95, 110]
        elif phoneme == "nfr":
            keys = ["n", "f", "r"]
            durs = [0.18/speed, 0.14/speed, 0.24/speed]
            f0s = [108, 120, 100]
        elif phoneme == "ḫpr":
            keys = ["ḫ", "p", "r"]
            durs = [0.16/speed, 0.12/speed, 0.26/speed]
            f0s = [115, 125, 105]
        elif phoneme == "snb":
            keys = ["s", "n", "b"]
            durs = [0.13/speed, 0.18/speed, 0.24/speed]
            f0s = [120, 110, 100]
        elif phoneme == "wbn":
            keys = ["w", "b", "n"]
            durs = [0.20/speed, 0.16/speed, 0.22/speed]
            f0s = [95, 100, 90]
        
        # Generate with variations
        audio = coart(keys, durs, f0s, 24)
        audio = rev(audio, 0.48, reverb_mix)
        
        # Add effects based on variation name
        if "dark" in var_name or "mystic" in var_name:
            # Add lowpass filter
            b, a = butter(2, 3000/(SR/2), btype='low')
            audio = lfilter(b, a, audio)
        elif "bright" in var_name or "sun" in var_name:
            # Add highpass filter
            b, a = butter(2, 200/(SR/2), btype='high')
            audio = lfilter(b, a, audio)
        
        write(f"{OUT}/{phoneme}_{var_name}.wav", audio)

print("\n=== Generating Tempo Variations ===")
# Generate tempo variations for all biliterals
tempos = [0.7, 0.85, 1.0, 1.15, 1.3]
biliteral_sequences = {
    "mn": (["m", "n"], [0.22, 0.28], [105, 98]),
    "ḥr": (["ḥ", "r"], [0.20, 0.25], [110, 102]),
    "sḏ": (["s", "ḏ"], [0.16, 0.22], [125, 118]),
    "ṯꜣ": (["ṯ", "ꜣ"], [0.18, 0.30], [115, 95])
}

for seq_name, (keys, durs, f0s) in biliteral_sequences.items():
    for tempo in tempos:
        # Adjust durations for tempo
        new_durs = [d / tempo for d in durs]
        audio = coart(keys, new_durs, f0s, 22)
        audio = rev(audio, mix=0.28)
        write(f"{OUT}/biliteral_{seq_name}_tempo_{int(tempo*100)}.wav", audio)

print("\n=== Generating Acoustic Treatments ===")
# Generate different acoustic treatments for the Oracle phrase
oracle_segments = []
for idx, dur, f0 in zip([0, 6, 1, 24, 4, 5, 22, 0], 
                         [0.55, 0.45, 0.35, 0.60, 0.50, 0.45, 0.40, 0.70],
                         [85, 92, 88, 78, 82, 80, 95, 75]):
    seg_audio = seg(phoneme_keys[idx], f0=f0, dur=dur)
    oracle_segments.append(seg_audio)
    oracle_segments.append(np.zeros(int(0.08 * SR), dtype=np.float32))

oracle = np.concatenate(oracle_segments)

# Different reverb treatments
treatments = [
    ("temple_light", 0.3, 0.2),
    ("temple_medium", 0.45, 0.35),
    ("temple_heavy", 0.6, 0.5),
    ("cathedral", 0.7, 0.6),
    ("cave", 0.75, 0.65),
    ("dry", 0.15, 0.1)
]

for treatment_name, decay, mix in treatments:
    treated = rev(oracle, decay, mix)
    fade = int(0.15 * SR)
    treated[:fade] *= np.linspace(0, 1, fade)
    treated[-fade:] *= np.linspace(1, 0, fade)
    write(f"{OUT}/oracle_{treatment_name}.wav", treated)

print("\nDone - Variations generated.")
